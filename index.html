let adicionaisEscolhidos = []; function abrirDetalhes(id) { const p = DATA.items.find(item => item.id === id); qtdModal = 1; adicionaisEscolhidos = []; // Reseta let htmlAdicionais = ""; if (p.adicionais && p.adicionais.length > 0) { htmlAdicionais = `

Turbine seu pedido:

${p.adicionais.map((ad, idx) => ` `).join('')}
`; } document.getElementById('detalhe-conteudo').innerHTML = `

${p.nome}

${p.descricao || ''}

${htmlAdicionais}
`; document.getElementById('modal-produto').classList.remove('hidden'); } function toggleAdicional(nome, preco, precoBase) { const idx = adicionaisEscolhidos.findIndex(a => a.nome === nome); if (idx > -1) adicionaisEscolhidos.splice(idx, 1); else adicionaisEscolhidos.push({ nome, preco }); // Atualiza o preço no botão const totalAdicionais = adicionaisEscolhidos.reduce((acc, a) => acc + a.preco, 0); document.getElementById('sub-mod').innerText = (precoBase + totalAdicionais).toFixed(2); } function finalizarSelecao(nomeOriginal, id) { const p = DATA.items.find(item => item.id === id); const totalAdicionais = adicionaisEscolhidos.reduce((acc, a) => acc + a.preco, 0); const nomeCompleto = adicionaisEscolhidos.length > 0 ? `${nomeOriginal} (+ ${adicionaisEscolhidos.map(a => a.nome).join(', ')})` : nomeOriginal; addCar(nomeCompleto, p.preco + totalAdicionais, 1); fecharModal(); }

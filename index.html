// Variável global para controlar a quantidade no modal let qtdModal = 1; function abrirDetalhes(id) { const p = DATA.items.find(item => item.id === id); qtdModal = 1; // Reseta para 1 toda vez que abre const conteudo = document.getElementById('detalhe-conteudo'); conteudo.innerHTML = `

${p.nome}

R$ ${p.preco.toFixed(2)}

${p.descricao || 'Sem descrição.'}

1
`; document.getElementById('modal-produto').classList.remove('hidden'); } function alterarQtdModal(valor) { const pPreco = parseFloat(document.getElementById('subtotal-modal').innerText) / qtdModal; // Pega o preço unitário qtdModal += valor; if (qtdModal < 1) qtdModal = 1; document.getElementById('display-qtd-modal').innerText = qtdModal; document.getElementById('subtotal-modal').innerText = (pPreco * qtdModal).toFixed(2); }
